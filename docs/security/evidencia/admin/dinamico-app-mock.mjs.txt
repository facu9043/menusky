async function login(email) {
  const jar = new Map();
  const s = createServerClient(MOCK, ANON, { cookies: { getAll: () => [...jar].map(([name, value]) => ({ name, value })), setAll: (l) => l.forEach(({ name, value }) => jar.set(name, value)) } });
  const { data, error } = await s.auth.signInWithPassword({ email, password: "demo-1234" });
  if (error) throw error;
  return { cookie: [...jar].map(([k, v]) => `${k}=${v}`).join("; "), token: data.session.access_token };
}

  return `${r.status} ${t.length > 160 ? t.slice(0, 160) + "...(" + t.length + " bytes)" : t}`;
};
